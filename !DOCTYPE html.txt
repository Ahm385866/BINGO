function haptic(type='light'){try{sdk.hapticFeedback(type)}catch(e){}}

let audioCtx=null,musicTimer=null,musicMuted=false,musicStep=0,musicNodes=[];
// Short, clearly audible "dattiri-dattat" arcade melody.
const melody=[{n:392,t:0.18},{n:392,t:0.18},{n:523,t:0.28},{n:440,t:0.18},{n:440,t:0.18},{n:587,t:0.28},{n:523,t:0.18},{n:440,t:0.18},{n:392,t:0.36},{n:330,t:0.18},{n:392,t:0.18},{n:440,t:0.3}];
async function startMusic(){
  if(musicMuted)return;
  try{
    const AudioCtor=window.AudioContext||window.webkitAudioContext;
    if(!AudioCtor)return;
    audioCtx=audioCtx||new AudioCtor();
    if(audioCtx.state==='suspended')await audioCtx.resume();
    clearTimeout(musicTimer); musicStep=0;
    const playNote=()=>{
      if(musicMuted||!audioCtx||audioCtx.state!=='running')return;
      const beat=melody[musicStep%melody.length],now=audioCtx.currentTime;
      const osc=audioCtx.createOscillator(),gain=audioCtx.createGain();
      const bass=audioCtx.createOscillator(),bassGain=audioCtx.createGain();
      osc.type='square'; osc.frequency.setValueAtTime(beat.n,now);
      bass.type='triangle'; bass.frequency.setValueAtTime(beat.n/2,now);
      gain.gain.setValueAtTime(0.0001,now);
      gain.gain.linearRampToValueAtTime(0.28,now+0.018);
      gain.gain.exponentialRampToValueAtTime(0.0001,now+beat.t-0.02);
      bassGain.gain.setValueAtTime(0.0001,now);
      bassGain.gain.linearRampToValueAtTime(0.11,now+0.02);
      bassGain.gain.exponentialRampToValueAtTime(0.0001,now+beat.t-0.03);
      osc.connect(gain);gain.connect(audioCtx.destination);
      bass.connect(bassGain);bassGain.connect(audioCtx.destination);
      osc.start(now);bass.start(now);osc.stop(now+beat.t);bass.stop(now+beat.t);
      musicNodes.push(osc,bass);
      musicStep++;
      musicTimer=setTimeout(playNote,beat.t*1000);
    };
    playNote();
  }catch(e){musicTimer=null;}
}
function stopMusic(){
  clearTimeout(musicTimer);musicTimer=null;
  musicNodes.forEach(node=>{try{node.stop()}catch(e){}});musicNodes=[];
  if(audioCtx&&audioCtx.state==='running')audioCtx.suspend().catch(()=>{});
}

function generateCardNumbers(){
  const nums=[];
  while(nums.length<15){
    const n=Math.floor(Math.random()*TOTAL_NUMBERS)+1;
    if(!nums.includes(n))nums.push(n);
  }
  return nums.sort((a,b)=>a-b);
}

function isRowCompleted(p,rowIndex){
  const rowNumbers=p.numbers.slice(rowIndex*5,rowIndex*5+5);
  return rowNumbers.length===5&&rowNumbers.every(n=>p.marked.includes(n));
}

function completedRows(p){
  let count=0;
  for(let r=0;r<3;r++){
    if(isRowCompleted(p,r))count++;
  }
  return count;
}

function activePlayers(){return players.filter(p=>!p.eliminated)}

function renderCards(){
  const container=document.getElementById('playersContainer');
  container.innerHTML='';
  activePlayers().forEach(p=>{
    if(p.isUser&&p.numbers.length!==15)p.numbers=generateCardNumbers();
    const rows=completedRows(p),box=document.createElement('div');
    box.className='card-box '+(p.isUser?'user-card ':'')+(p.isWinner?'winner ':'')+(rows>p.announced?'milestone':'');
    box.id='card-'+p.id;
    const header=document.createElement('div');
    header.className='card-header';
    header.innerHTML='<span>'+p.name+' - <b class="money">$'+p.money+'</b></span>';
    box.appendChild(header);
    const grid=document.createElement('div');
    grid.className='bingo-grid';
    p.numbers.forEach((n,i)=>{
      const cell=document.createElement('div');
      const rowIndex=Math.floor(i/5);
      const rowDone=isRowCompleted(p,rowIndex);
      cell.className='cell '+(p.marked.includes(n)?'marked ':'')+(rowDone?'milestone-cell':'');
      cell.textContent=String(n);
      grid.appendChild(cell);
    });
    box.appendChild(grid);
    container.appendChild(box);
  });
}

function payPrize(winner,perCardAmount){
  const contributors=activePlayers().filter(p=>p!==winner);
  animateMoneyTransfer(winner,perCardAmount,contributors);
  contributors.forEach(p=>{p.money-=perCardAmount});
  winner.money+=perCardAmount*contributors.length;
  const eliminated=contributors.filter(p=>p.money<=0);
  eliminated.forEach(p=>{p.money=0;p.eliminated=true});
  renderCards();
  return {contributors,eliminated};
}

function handlePostPayoutState(eliminated,prefix){
  if(players[0].eliminated){
    endGame(prefix+' Your card ran out of money. You are eliminated! Game over.',true);
    return true;
  }
  if(activePlayers().length<=1){
    endGame(prefix+' All other opponents were eliminated. Game over! Final Profit: $'+currentScore(),false);
    return true;
  }
  return false;
}

function checkWinner(){
  if(gameEnded)return;

  // 1. BINGO Check
  const winner=players.find(p=>!p.eliminated&&completedRows(p)===3);
  if(winner){
    gameActive=false;
    paused=true;
    winner.isWinner=true;
    const {contributors,eliminated}=payPrize(winner,300);
    
    let countdown=5;
    const msgPrefix=winner.name+' got BINGO! Won $300 prize per card from '+contributors.length+' card(s).'+eliminationNote(eliminated);
    
    if(handlePostPayoutState(eliminated,msgPrefix))return;

    document.getElementById('drawBtn').disabled=true;
    const statusElem=document.getElementById('gameStatus');
    statusElem.textContent=msgPrefix+' Transitioning to choices... ('+countdown+'s)';
    haptic('success');

    const interval=setInterval(()=>{
      countdown--;
      if(countdown>0){
        statusElem.textContent=msgPrefix+' Transitioning to choices... ('+countdown+'s)';
      }else{
        clearInterval(interval);
        showTombalaChoice(winner);
      }
    },1000);

    return;
  }

  // 2. 2ND LINE Check
  if(!secondCinkoPaid){
    const secondCandidate=players.find(p=>!p.eliminated&&completedRows(p)>=2);
    if(secondCandidate){
      firstCinkoPaid=true;
      secondCinkoPaid=true;
      secondCandidate.announced=Math.max(secondCandidate.announced,2);
      const {contributors,eliminated}=payPrize(secondCandidate,200);
      const totalAward=200*contributors.length;
      const msg=secondCandidate.name+' completed 2ND LINE! Awarded $'+totalAward+' to '+secondCandidate.name+'.'+eliminationNote(eliminated);
      if(handlePostPayoutState(eliminated,msg))return;
      paused=true;
      document.getElementById('drawBtn').textContent='Continue';
      document.getElementById('gameStatus').textContent=msg+' Press button to continue.';
      haptic('success');
      return;
    }
  }

  // 3. 1ST LINE Check
  if(!firstCinkoPaid){
    const firstCandidate=players.find(p=>!p.eliminated&&completedRows(p)>=1);
    if(firstCandidate){
      firstCinkoPaid=true;
      firstCandidate.announced=Math.max(firstCandidate.announced,1);
      const {contributors,eliminated}=payPrize(firstCandidate,100);
      const totalAward=100*contributors.length;
      const msg=firstCandidate.name+' completed 1ST LINE! Awarded $'+totalAward+' to '+firstCandidate.name+'.'+eliminationNote(eliminated);
      if(handlePostPayoutState(eliminated,msg))return;
      paused=true;
      document.getElementById('drawBtn').textContent='Continue';
      document.getElementById('gameStatus').textContent=msg+' Press button to continue.';
      haptic('success');
      return;
    }
  }

  if(!bag.length){
    endGame('Game ended in a tie! Final Profit: $'+currentScore(),false);
  }
}

function renderStartCards(){
  const area=document.getElementById('startCards');
  area.innerHTML='';
  startChoices.forEach((card,index)=>{
    const box=document.createElement('div');
    box.className='start-card'+(index===0?' selected':'');
    box.addEventListener('click',()=>{
      selectedStartCard=card.slice();
      document.querySelectorAll('.start-card').forEach((el,i)=>el.classList.toggle('selected',i===index));
      haptic('light');
    });
    const title=document.createElement('strong');
    title.textContent='Card '+(index+1);
    box.appendChild(title);
    const grid=document.createElement('div');
    grid.className='start-grid';
    card.forEach(n=>{
      const cell=document.createElement('div');
      cell.className='start-cell';
      cell.textContent=n;
      grid.appendChild(cell);
    });
    box.appendChild(grid);
    area.appendChild(box);
  });
  selectedStartCard=startChoices[0].slice();
}
